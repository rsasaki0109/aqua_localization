cmake_minimum_required(VERSION 3.16)
project(mbes_offline LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rclcpp REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters registration)
find_package(aqua_sonar_loc REQUIRED)
add_executable(mbes-offline ../mbes_offline.cpp)
target_compile_features(mbes-offline PRIVATE cxx_std_17)
target_link_libraries(mbes-offline aqua_sonar_loc::scan_matcher)
ament_target_dependencies(mbes-offline sensor_msgs geometry_msgs pcl_conversions)
